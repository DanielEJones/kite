def main(x: num): num =
  let initial := mk_state(
    mk_vec2(0, 0),
    mk_vec2(1, 0),
  );

  let final := initial.apply_n_times(10, 
    .update_player_pos(
      .update_x(.add(1))
      .update_y(.add(1))
    )
    .update_player_vel(
      .update_x(.mul(0.9))
      .update_y(.mul(0.9))
    )
  );

  final.get_player_vel.get_x;


type state = | mk_state(
  vec2, vec2,
);

def get_player_pos(s: state): vec2 =
  match s
  \mk_state(player_pos, _) => player_pos;

def with_player_pos(s: state, pos: vec2): state =
  match s
  \mk_state(_, player_vel) => mk_state(pos, player_vel);

def update_player_pos(s: state, f: vec2 -> vec2): state =
  s.with_player_pos(f(s.get_player_pos));

def get_player_vel(s: state): vec2 =
  match s
  \mk_state(_, player_vel) => player_vel;

def with_player_vel(s: state, vel: vec2): state =
  match s
  \mk_state(player_pos, _) => mk_state(player_pos, vel);

def update_player_vel(s: state, f: vec2 -> vec2): state = 
  s.with_player_vel(f(s.get_player_vel));


type vec2 = | mk_vec2(
  num, 
  num,
);

def get_x(v: vec2): num = 
  match v
  \mk_vec2(x, _) => x;

def with_x(v: vec2, x: num): vec2 = 
  match v
  \mk_vec2(_, y) => mk_vec2(x, y);

def update_x(v: vec2, f: num -> num): vec2 = 
  v.with_x(f(v.get_x));

def get_y(v: vec2): num = 
  match v
  \mk_vec2(_, y) => y;

def with_y(v: vec2, y: num): vec2 =
  match v
  \mk_vec2(x, _) => mk_vec2(x, y);

def update_y(v: vec2, f: num -> num): vec2 = 
  v.with_y(f(v.get_y));



def apply_n_times(@a: type, acc: a, n: num, step: a -> a): a =
  if n < 1
    then acc
    else step(acc).apply_n_times(n - 1, step);

def add(a: num, b: num): num = a + b;
def mul(a: num, b: num): num = a * b;
